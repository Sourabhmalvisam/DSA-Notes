    import java.util.*;

public class Solution { // Use NavigableSet for TreeSet.
    public static void main(String[] args) {

        NavigableSet<Integer> navigableSet = new TreeSet<>();
        navigableSet.addAll(Arrays.asList(0, 1, 2, 3, 4, 5, 6));

        System.out.println("Head set (exclusive) of 3: " + navigableSet.headSet(3, false));
        System.out.println("Tail set (exclusive) of 3: " + navigableSet.tailSet(3, false));
        System.out.println();
        System.out.println("Head set (inclusive) of 3: " + navigableSet.headSet(3, true));
        System.out.println("Tail set (inclusive) of 3: " + navigableSet.tailSet(3, true));

        System.out.println();
        System.out.println("Sub set (exclusive) from 1 to 5: " + navigableSet.subSet(1, false, 5, false));
        System.out.println("Sub set (inclusive) from 1 to 5: " + navigableSet.subSet(1, true, 5, true));
        System.out.println();

        System.out.println("Ceiling of 3: " + navigableSet.ceiling(3));
        System.out.println("Floor of 3: " + navigableSet.floor(3));
        System.out.println("Higher than 3: " + navigableSet.higher(3));
        System.out.println("Lower than 3: " + navigableSet.lower(3));
        System.out.println();

        NavigableSet<Integer> descendingSet = navigableSet.descendingSet();
        System.out.println("Descending set: " + descendingSet);

        System.out.println();
        System.out.println("Poll first element: " + navigableSet.pollFirst());
        System.out.println("Poll last element: " + navigableSet.pollLast());
    }
}

    import java.util.NavigableMap;
import java.util.TreeMap;

public class NavigableMapExample {
    public static void main(String[] args) {
        NavigableMap<Integer, String> navigableMap = new TreeMap<>();

        navigableMap.put(0, "Zero");
        navigableMap.put(1, "One");
        navigableMap.put(2, "Two");
        navigableMap.put(3, "Three");
        navigableMap.put(4, "Four");
        navigableMap.put(5, "Five");
        navigableMap.put(6, "Six");

        System.out.println("HeadMap(3, true): " + navigableMap.headMap(3, true)); // {0=Zero, 1=One, 2=Two, 3=Three}
        System.out.println("TailMap(3, true): " + navigableMap.tailMap(3, true)); // {3=Three, 4=Four, 5=Five, 6=Six}
        System.out.println();
        System.out.println("HeadMap(3, false): " + navigableMap.headMap(3, false)); // {0=Zero, 1=One, 2=Two}
        System.out.println("TailMap(3, false): " + navigableMap.tailMap(3, false)); // {4=Four, 5=Five, 6=Six}
        System.out.println();
        System.out.println("SubMap(1, true, 5, true): " + navigableMap.subMap(1, true, 5, true));
        // {1=One, 2=Two,3=Three, 4=Four, 5=Five}
        System.out.println("SubMap(1, false, 5, false): " + navigableMap.subMap(1, false, 5, false));
        // {2=Two, 3=Three, 4=Four}
        System.out.println();

        System.out.println("Ceiling Entry(3): " + navigableMap.ceilingEntry(3)); // 3=Three
        System.out.println("Floor Entry(3): " + navigableMap.floorEntry(3)); // 3=Three
        System.out.println("Higher Entry(3): " + navigableMap.higherEntry(3)); // 4=Four
        System.out.println("Lower Entry(3): " + navigableMap.lowerEntry(3)); // 2=Two

        System.out.println();
        System.out.println("First Entry: " + navigableMap.firstEntry()); // 0=Zero
        System.out.println("Last Entry: " + navigableMap.lastEntry()); // 6=Six
        System.out.println("Poll First Entry: " + navigableMap.pollFirstEntry()); // 0=Zero
        System.out.println("Poll Last Entry: " + navigableMap.pollLastEntry()); // 6=Six

        System.out.println();
        System.out.println("Descending Map: " + navigableMap.descendingMap());
        // {5=Five, 4=Four, 3=Three, 2=Two, 1=One}

    }
}
