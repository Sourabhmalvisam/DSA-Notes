    class Solution { //DSAQuickRevision
    public boolean search(int[] nums, int target) {
        int n = nums.length;
        int low = 0, high = n - 1;
        while (low <= high) {
            int mid = (low + high) / 2;

            // Check if mid points to the target
            if (nums[mid] == target)
                return true;

            // Handle duplicates by adjusting boundaries
            if (nums[low] == nums[mid] && nums[mid] == nums[high]) {
                low++;
                high--;
                continue;
            }

            // If left part is sorted
            if (nums[low] <= nums[mid]) {
                if (nums[low] <= target && target <= nums[mid]) {
                    // Target is in the left sorted part
                    high = mid - 1;
                } else {
                    // Target is not in the left sorted part
                    low = mid + 1;
                }
            } else { 
                // If right part is sorted
                if (nums[mid] <= target && target <= nums[high]) {
                    // Target is in the right sorted part
                    low = mid + 1;
                } else {
                    // Target is not in the right sorted part
                    high = mid - 1;
                }
            }
        }
        return false;
    }
}

    class Solution {  // DSAQuickRevision
    public double median(int[] nums1, int[] nums2) {
        int n1 = nums1.length, n2 = nums2.length;

        // Always binary search on smaller array for log(min(n1, n2)) complexity
        if (n1 > n2)
            return median(nums2, nums1); // Ensure nums1 is smaller

        int left = (n1 + n2 + 1) / 2; // Total elements in left half
        int low = 0, high = n1;

        while (low <= high) {
            int mid1 = (low + high) / 2;       // Partition in nums1
            int mid2 = left - mid1;            // Remaining in nums2

            // Edge values for comparison (handle out-of-bound with ±∞)
            int l1 = (mid1 > 0) ? nums1[mid1 - 1] : Integer.MIN_VALUE;
            int l2 = (mid2 > 0) ? nums2[mid2 - 1] : Integer.MIN_VALUE;
            int r1 = (mid1 < n1) ? nums1[mid1] : Integer.MAX_VALUE;
            int r2 = (mid2 < n2) ? nums2[mid2] : Integer.MAX_VALUE;

            // Check if correct partition
            if (l1 <= r2 && l2 <= r1) {
                // Odd total length → max of left parts
                if ((n1 + n2) % 2 == 1) 
                    return Math.max(l1, l2);
                // Even total length → average of middle two
                return (Math.max(l1, l2) + Math.min(r1, r2)) / 2.0;
            } 
            else if (l1 > r2) 
                high = mid1 - 1; // Too many from nums1 → move left
            else 
                low = mid1 + 1;  // Too few from nums1 → move right
        }
        return 0; // Should never reach here (valid input assumed)
    }
}

    class Solution {
    //Function to find the median of two sorted arrays.
    public double median(int[] arr1, int[] arr2) {
        // Size of two given arrays
        int n1 = arr1.length, n2 = arr2.length;

        int[] merged = new int[n1 + n2];
        // Apply the merge step
        int i = 0, j = 0, k = 0;
        while (i < n1 && j < n2) {
            if (arr1[i] < arr2[j]) merged[k++] = arr1[i++];
            else merged[k++] = arr2[j++];
        }

        // Copy the remaining elements
        while (i < n1) merged[k++] = arr1[i++];
        while (j < n2) merged[k++] = arr2[j++];

        // Find the median
        int n = n1 + n2;
        if (n % 2 == 1) {
            return (double) merged[n / 2];
        }

        double median = ((double) merged[n / 2] + (double) merged[(n / 2) - 1]) / 2.0;
        return median;
    }
}

        int first = -1;

        // Applying Binary Search Algorithm
        while(low <= high) {
            int mid = low + (high - low) / 2;

            /*  If the target element is found, we 
                update the first variable to mid and
                eliminate the right half to look for 
                more smaller index where target is present */
            if(nums[mid] == target) {
                first = mid;
                high = mid - 1;  
            } 

            /*  If middle element is smaller,
                we eliminate the left half  */
            else if(nums[mid] < target) {
                low = mid + 1;  
            } 
            
            /*  If middle element is greater,
                we eliminate the right half  */

                high = mid - 1;  
            }
        }

    public static int findPeakElement(int[] arr) {
        int n = arr.length; // Size of array

        // Edge cases:
        if (n == 1)
            return 0;
        if (arr[0] > arr[1])
            return 0;
        if (arr[n - 1] > arr[n - 2])
            return n - 1;

        int low = 1, high = n - 2;
        while (low <= high) {
            int mid = (low + high) / 2;

            // If arr[mid] is the peak:
            if (arr[mid - 1] < arr[mid] && arr[mid] > arr[mid + 1])
                return mid;

            // If we are in the left:
            else if (arr[mid] > arr[mid - 1])
                low = mid + 1;

            // If we are in the right:
            else
                high = mid - 1;
        }
        return -1;
    }

    public int smallestDivisor(int[] arr, int k) {
        int max = -1;

        // Find the maximum value in the array
        for (int val : arr) {
            max = Math.max(val, max);
        }

        int low = 1;
        int high = max;
        int ans = -1;

        // Binary search for the smallest possible divisor
        while (low <= high) {
            int mid = (low + high) / 2;

            if (isPossible(arr, mid, k)) {
                ans = mid;
                high = mid - 1; // Try smaller divisor
            } else {
                low = mid + 1;  // Try larger divisor
            }
        }

        return ans;
    }

        int sum = 0;

        for (int val : arr) {
            sum += Math.ceil((double) val / divisor);
        }

        return sum <= limit;
    }

    public static int aggressiveCows(int[] stalls, int k) {
        int n = stalls.length;
        Arrays.sort(stalls);
        int low = 1; // Minimum possible distance
        int high = stalls[n - 1] - stalls[0]; // Maximum possible distance
        int result = -1; // Variable to store the final answer

        // Binary search to find the maximum minimum distance
        while (low <= high) {
            int mid = low + (high - low) / 2;

            // Check if we can place all cows with the current minimum distance
            if (canWePlaceAll(stalls, mid, k)) {
                low = mid + 1;
                result = mid; // Update result with the current minimum distance
            } else {
                high = mid - 1;
            }
        }

        return result;
    }

    public static boolean canWePlaceAll(int[] stalls, int distance, int k) {
        int lastCowAt = stalls[0];
        int count = 1;

        // Iterate through stalls to check if we can place all cows
        for (int i = 1; i < stalls.length; i++) {
            if (stalls[i] - lastCowAt >= distance) {
                count++;
                lastCowAt = stalls[i];
            }
        }

        return count >= k; // Return true if we can place all cows
    }
}

    class Solution {
    public int minDaysBloom(int[] arr, int k, int m) {
        if ((long) m * k > arr.length) return -1; // Not enough flowers
        
        int low = Integer.MAX_VALUE, high = Integer.MIN_VALUE;
        for (int day : arr) {
            low = Math.min(low, day);
            high = Math.max(high, day);
        }

        int ans = -1;
        while (low <= high) {
            int mid = low + (high - low) / 2;
            if (canMakeBouquets(arr, k, m, mid)) {
                ans = mid;
                high = mid - 1; // try smaller day
            } else {
                low = mid + 1;
            }
        }
        return ans;
    }

    private boolean canMakeBouquets(int[] arr, int k, int m, int day) {
        int flowers = 0;
        int bouquets = 0;

        for (int bloomDay : arr) {
            if (bloomDay <= day) {
                flowers++;
                if (flowers == k) {
                    bouquets++;
                    flowers = 0;
                }
            } else {
                flowers = 0;
            }
            if (bouquets >= m) return true;
        }
        return false;
    }
}

        while (low <= high) {
            int mid = (low + high) / 2;

            // If nums[mid] is the single element:
            if (nums[mid] != nums[mid + 1] && nums[mid] != nums[mid - 1]) {
                return nums[mid];
            }

            // We are in the left part:
            if ((mid % 2 == 1 && nums[mid] == nums[mid - 1])
                || (mid % 2 == 0 && nums[mid] == nums[mid + 1])) {
                // Eliminate the left half:
                low = mid + 1;
            }
            // We are in the right part:
            else {
                // Eliminate the right half:
                high = mid - 1;
            }
        }

        // Dummy return statement:
        return -1;
    }
}
  • Minimize Maximum Distance between Gas Stations
    class Solution { //DSAStriver
    public static double minimiseMaxDistance(int arr[], int k) {
        int n = arr.length; // size of the array
        double low = 0;
        double high = arr[n - 1] - arr[0]; // Difference between the first and last positions

        double diff = 1e-6; // Precision for binary search
        double ans = -1; // Store the answer

        while (high - low > diff) {
            double mid = (low + high) / 2.0;
            int cnt = numberOfGasStationsRequired(mid, arr);
            if (cnt > k) {
                low = mid; // Increase the minimum distance
            } else {
                ans = mid; // Update the answer
                high = mid; // Decrease the maximum distance
            }
        }

        return ans; // Return the smallest maximum distance
    }

    public static int numberOfGasStationsRequired(double dist, int[] arr) {
        int n = arr.length; // Size of the array
        int cnt = 0; // Count of gas stations required
        for (int i = 1; i < n; i++) {
            cnt += (int) Math.ceil((arr[i] - arr[i - 1]) / dist) - 1;
        }
        return cnt;
    }
}
  • Find Nth root of a number
    class Solution{ //DSAStriver
    /* Helper function to check the power of mid
    with respect to m Returns: 1 - if mid^n == m,
    0 - if mid^n < m and 2 - if mid^n > m*/ 
    private int func(int mid, int n, int m) {
        long ans = 1;
        for (int i = 1; i <= n; i++) {
            ans *= mid;
            if (ans > m) return 2;
        }
        if (ans == m) return 1;
        return 0;
    }

    /* Function to find the nth root
    of m using binary search*/
    public int NthRoot(int N, int M) {
        // Binary search on the answer space
        int low = 1, high = M;

        while (low <= high) {
            int mid = (low + high) / 2;
            int midN = func(mid, N, M);
            if (midN == 1) {
                // Return mid if mid^N == M
                return mid;
            } else if (midN == 0) {
                // Move to the right half if mid^N < M
                low = mid + 1;
            } else {
                // Move to the left half if mid^N > M
                high = mid - 1;
            }
        }
        // Return -1 if no nth root found
        return -1;
    }
}

    class Solution{ //DSAStriver
public int kthElement(int[] a, int[] b, int k) {
    int m = a.length;
    int n = b.length;

    // Ensure a is smaller array for optimization
    if (m > n) {
        // Swap a and b
        return kthElement(b, a, k); 
    }
    
    // Length of the left half
    int left = k; 

    // Apply binary search
    int low = Math.max(0, k - n), high = Math.min(k, m);
    while (low <= high) {
        int mid1 = (low + high) >> 1;
        int mid2 = left - mid1;

        // Initialize l1, l2, r1, r2
        int l1 = (mid1 > 0) ? a[mid1 - 1] : Integer.MIN_VALUE;
        int l2 = (mid2 > 0) ? b[mid2 - 1] : Integer.MIN_VALUE;
        int r1 = (mid1 < m) ? a[mid1] : Integer.MAX_VALUE;
        int r2 = (mid2 < n) ? b[mid2] : Integer.MAX_VALUE;

        // Check if we have found the answer
        if (l1 <= r2 && l2 <= r1) {
            return Math.max(l1, l2);
        } 
        else if (l1 > r2) {
            // Eliminate the right half
            high = mid1 - 1;
        } 
        else {
            // Eliminate the left half
            low = mid1 + 1;
        }
    }
    
     // Dummy return statement 
    return -1;
}

    class Solution{ //DSAStriver

    with the maximum element in a given column*/
    public int maxElement(int[][] arr, int col) {
        int n = arr.length;
        int max = Integer.MIN_VALUE;
        int index = -1;
        
        /* Iterate through each row to find the
        maximum element in the specified column*/
        for (int i = 0; i < n; i++) {
            if (arr[i][col] > max) {
                max = arr[i][col];
                index = i;
            }
        }
        //Return the index
        return index; 
    }
    
    /* Function to find a peak element in 
    the 2D matrix using binary search */
    public int[] findPeakGrid(int[][] arr) {
        int n = arr.length;   
        int m = arr[0].length; 
        
        /* Initialize the lower bound for 
        and upper bound for binary search */
        int low = 0;           
        int high = m - 1;      
        
        // Perform binary search on columns
        while (low <= high) {
            int mid = (low + high) / 2;  
            
            /* Find the index of the row with the 
            maximum element in the middle column*/
            int row = maxElement(arr, mid);
            
            /* Determine the elements to left and 
            right of middle element in the found row */
            int left = mid - 1 >= 0 ? arr[row][mid - 1] : Integer.MIN_VALUE;
            int right = mid + 1 < m ? arr[row][mid + 1] : Integer.MIN_VALUE;
            
            /* Check if the middle element 
            is greater than its neighbors */
            if (arr[row][mid] > left && arr[row][mid] > right) {
                return new int[]{row,mid};  
            } 
            else if (left > arr[row][mid]) {
                high = mid - 1;  
            } 
            else {
                low = mid + 1;
            }
        }
        
        // Return -1 if no peak element is found
        return new int[]{-1,-1};  
    }
