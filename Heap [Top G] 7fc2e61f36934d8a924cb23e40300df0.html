    class Solution {
    public List<String> topKFrequent(String[] words, int k) {
        Map<String, Integer> freqMap = new HashMap<>();

        // Step 1: Count frequency
        for (String word : words) {
            freqMap.put(word, freqMap.getOrDefault(word, 0) + 1);
        }

        // Step 2: Use min-heap with custom comparator
        PriorityQueue<String> heap = new PriorityQueue<>((w1, w2) -> {
            int freqCompare = freqMap.get(w1) - freqMap.get(w2);
            if (freqCompare == 0) {
                // Reverse lexicographical order
                return w2.compareTo(w1); // Imp: isko reverse compare kyu kiya?
            }
            return freqCompare;
        });

        // Step 3: Add to heap, maintain size k
        for (String word : freqMap.keySet()) { // Imp: yaha map hi iterate krna, wrna wrong ans
            heap.add(word); //Imp: Isko if ke niche kyu nahi likha?
            if (heap.size() > k) {
                heap.poll(); // remove least frequent
            }
        }

        // Step 4: Extract from heap (in reverse)
        List<String> result = new LinkedList<>();
        while (!heap.isEmpty()) {
            result.add(0, heap.poll()); // reverse order
        }

        return result;
    }
}

    class KthLargest { //DSAQuickRevision
    int k; // Store the kth position
    PriorityQueue<Integer> pq; // Min-heap to store k largest elements

    public KthLargest(int k, int[] nums) {
        this.k = k;
        pq = new PriorityQueue<>(); // Default is min-heap

        // Add all elements using the add() method to maintain k largest
        for (int val : nums) {
            add(val);
        }
    }

    public int add(int val) {
        pq.add(val); // Add the new element

        // If heap size exceeds k, remove the smallest (not in top k)
        if (pq.size() > k) {
            pq.poll();
        }

        // Top of min-heap is kth largest
        return pq.peek();
    }
}

    class Solution { //DSAQuickRevision

    // Main function to sort an array using Heap Sort
    public void heapSort(int[] array) {
        int n = array.length;

        // Step 1: Build a Max-Heap
        for (int i = n / 2 - 1; i >= 0; i--) {
            heapify(array, n, i);
        }

        // Step 2: Extract elements from the heap one by one
        for (int i = n - 1; i >= 0; i--) {
            // Swap the root (maximum element) with the last element
            swap(array, 0, i);

            // Restore the Max-Heap property for the reduced heap
            heapify(array, i, 0);
        }
    }

    // Function to maintain the Max-Heap property
    private void heapify(int[] array, int n, int i) {
        int largest = i;        // Assume current node is the largest
        int left = 2 * i + 1;   // Left child
        int right = 2 * i + 2;  // Right child

        // Check if the left child is larger than the current node
        if (left < n && array[left] > array[largest]) {
            largest = left;
        }

        // Check if the right child is larger than the largest so far
        if (right < n && array[right] > array[largest]) {
            largest = right;
        }

        // If the largest is not the current node, swap and recurse
        if (largest != i) {
            swap(array, i, largest);
            heapify(array, n, largest);
        }
    }

    // Utility function to swap two elements in an array
    private void swap(int[] array, int i, int j) {
        int temp = array[i];
        array[i] = array[j];
        array[j] = temp;
    }
}
  • Heapify Algorithm
    class Solution { //DSAQuickRevision
    
    // Swap two elements in the array
    private void swap(int[] arr, int i, int j) {
        int temp = arr[i];
        arr[i] = arr[j];
        arr[j] = temp;
    }

    // Heapify downwards to maintain min-heap property
    private void heapifyDown(int[] arr, int ind) {
        int n = arr.length;
        int smallest = ind;
        int left = 2 * ind + 1;
        int right = 2 * ind + 2;

        if (left < n && arr[left] < arr[smallest]) {
            smallest = left;
        }
        if (right < n && arr[right] < arr[smallest]) {
            smallest = right;
        }

        if (smallest != ind) {
            swap(arr, ind, smallest);
            heapifyDown(arr, smallest);
        }
    }

    // Heapify upwards to maintain min-heap property
    private void heapifyUp(int[] arr, int ind) {
        int parent = (ind - 1) / 2;

        if (arr[ind] < arr[parent]) {
            swap(arr, ind, parent);
            heapifyUp(arr, parent);
        }
    }

    // Update value at index and restore min-heap
    public void heapify(int[] nums, int ind, int val) {
        if (nums[ind] > val) {
            nums[ind] = val;
            heapifyUp(nums, ind);
        } else {
            nums[ind] = val;
            heapifyDown(nums, ind);
        }
    }
}

    class MinHeap { //DSAStriver
    private ArrayList<Integer> heap; // Internal ArrayList to store heap elements

    // Constructor to initialize the heap
    public MinHeap() {
        heap = new ArrayList<>();
    }

    // Function to find the parent index of a given node
    private int parent(int index) {
        return (index - 1) / 2;
    }

    // Function to find the left child index of a given node
    private int leftChild(int index) {
        return 2 * index + 1;
    }

    // Function to find the right child index of a given node
    private int rightChild(int index) {
        return 2 * index + 2;
    }

    // Function to insert a new value into the heap
    public void insert(int value) {
        // Add the new value to the end of the heap
        heap.add(value);
        int currentIndex = heap.size() - 1; // Index of the newly added value

        // Bubble up to maintain the heap property
        while (currentIndex > 0 && heap.get(currentIndex) < heap.get(parent(currentIndex))) {
            swap(currentIndex, parent(currentIndex)); // Swap with parent if smaller
            currentIndex = parent(currentIndex); // Update the index to the parent's index
        }
    }

    // Function to extract the minimum value (root) from the heap
    public int extractMin() {
        if (heap.isEmpty()) {
            throw new IllegalStateException("Heap is empty!"); // Handle empty heap case
        }
        if (heap.size() == 1) {
            return heap.remove(0); // If only one element, remove and return it
        }

        int root = heap.get(0); // Minimum element (root of the heap)
        heap.set(0, heap.remove(heap.size() - 1)); // Replace root with the last element
        heapify(0); // Restore the heap property from the root
        return root;
    }

    // Function to restore the heap property by "heapifying" the subtree
    private void heapify(int index) {
        int smallest = index; // Assume the current index is the smallest
        int left = leftChild(index); // Left child index
        int right = rightChild(index); // Right child index

        // Check if the left child exists and is smaller than the current smallest
        if (left < heap.size() && heap.get(left) < heap.get(smallest)) {
            smallest = left;
        }
        // Check if the right child exists and is smaller than the current smallest
        if (right < heap.size() && heap.get(right) < heap.get(smallest)) {
            smallest = right;
        }
        // If the smallest is not the current index, swap and continue heapifying
        if (smallest != index) {
            swap(index, smallest);
            heapify(smallest); // Recursively heapify the affected subtree
        }
    }

    // Utility function to swap two elements in the heap
    private void swap(int i, int j) {
        int temp = heap.get(i);
        heap.set(i, heap.get(j));
        heap.set(j, temp);
    }

    // Function to print the current state of the heap
    public void printHeap() {
        System.out.println(heap);
    }
}

class Solution { //O(n)
    public void buildMinHeap(int[] nums) {
        int size = nums.length;
        // Heapify all non-leaf nodes (bottom-up)
        for (int i = (size / 2) - 1; i >= 0; i--) {
            heapify(nums, size, i);
        }
    }

    // Maintains Min-Heap property for subtree rooted at index i
    private void heapify(int[] heap, int size, int i) {
        int smallest = i;
        int left = 2 * i + 1;
        int right = 2 * i + 2;

        // Check left child
        if (left < size && heap[left] < heap[smallest]) {
            smallest = left;
        }

        // Check right child
        if (right < size && heap[right] < heap[smallest]) {
            smallest = right;
        }

        // If a smaller child exists, swap and heapify recursively
        if (smallest != i) {
            swap(heap, i, smallest);
            heapify(heap, size, smallest);
        }
    }

    // Swaps two elements in the array
    private void swap(int[] heap, int i, int j) {
        int temp = heap[i];
        heap[i] = heap[j];
        heap[j] = temp;
    }

    class Solution { //DSAStriver

    // Function to check if the given array is a min-heap
    public boolean isHeap(int[] nums) {
        int n = nums.length; // Size of the array
        
        // Iterate on the non-leaf nodes from the back
        for(int i = n/2 - 1; i >= 0; i--) {
            int leftChildInd = 2 * i + 1;
            int rightChildInd = 2 * i + 2;
            
            // If left child has a smaller value than the parent
            if(leftChildInd < n && nums[leftChildInd] < nums[i])
                return false;
                
            // If right child has a smaller value than parent
            if(rightChildInd < n && nums[rightChildInd] < nums[i])
                return false;
        }
        
        return true;
    }
}
  • Convert Min Heap to Max Heap
    class Solution { //DSAStriver
    
    // Function to recursively heapify the array downwards
    private void heapifyDown(int[] arr, int ind) {
        int n = arr.length; // Size of the array

        // To store the index of largest element
        int largestInd = ind;

        // Indices of the left and right children
        int leftChildInd = 2*ind + 1;
        int rightChildInd = 2*ind + 2;
        
        // If the left child holds larger value, update the largest index
        if(leftChildInd < n && arr[leftChildInd] > arr[largestInd]) {
            largestInd = leftChildInd;
        }

        // If the right child holds larger value, update the largest index
        if(rightChildInd < n && arr[rightChildInd] > arr[largestInd]) {
            largestInd = rightChildInd;
        }

        // If the largest element index is updated
        if(largestInd != ind) {
            // Swap the largest element with the current index
            int temp = arr[largestInd];
            arr[largestInd] = arr[ind];
            arr[ind] = temp;

            // Recursively heapify the lower subtree
            heapifyDown(arr, largestInd);
        }
    }

    // Function to convert a min-heap array to a max-heap array
    public int[] minToMaxHeap(int[] nums) {
        int n = nums.length;

        // Iterate backwards on the non-leaf nodes
        for(int i = n/2 - 1; i >= 0; i--) {
            // Heapify each node downwards to ensure max-heap property
            heapifyDown(nums, i);
        }
        return nums;
    }
