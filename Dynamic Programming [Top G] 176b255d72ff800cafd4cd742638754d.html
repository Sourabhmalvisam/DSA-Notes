DP (Memoization) Approach

  1. Try to represent the problem in terms of index or state

    → f(i), f(i, sum), f(i, j) etc.

  1. Do all possible stuffs on that index

    → Choices: pick / not pick, go left / right, cut / no cut, etc.

  1. sum of all stuffs → if question says "count all ways"

    min (of all stuffs) → if question says "find min cost / time / jumps"

    class Solution { //DSAQuickRevision
    // Helper function to solve the House Robber I problem using recursion and memoization
    public int helper(int[] nums, int ind, Integer[] dp) {
        if (ind == 0) {
            return nums[0]; // Base case: If at the first house, return its value.
        }
        if (ind < 0) return 0; // Base case: If out of bounds, return 0.

        if (dp[ind] != null) return dp[ind]; // Use the precomputed result if available.

        // Choose between picking the current house and skipping it:
        int pick = nums[ind] + helper(nums, ind - 2, dp); // If picked, skip the adjacent house.
        int notPick = helper(nums, ind - 1, dp); // If not picked, move to the previous house.

        // Store the maximum value of the two choices and return it.
        return dp[ind] = Math.max(pick, notPick);
    }

    // Main function to solve House Robber II problem
    public int rob(int[] nums) {
        int n = nums.length;
        
        // Edge cases
        if (n == 1) return nums[0];
        if (n == 2) return Math.max(nums[0], nums[1]);
        
        // Case 1: Rob houses from 0 to n-2 (ignore the last house)
        Integer[] dp1 = new Integer[n];
        int case1 = helper(Arrays.copyOfRange(nums, 0, n - 1), n - 2, dp1);
        
        // Case 2: Rob houses from 1 to n-1 (ignore the first house)
        Integer[] dp2 = new Integer[n];
        int case2 = helper(Arrays.copyOfRange(nums, 1, n), n - 2, dp2);
        
        // Return the maximum of the two cases
        return Math.max(case1, case2);
    }
}

    public int coinChange(int[] coins, int amount) {
        int n = coins.length;
        dp = new Integer[n][amount + 1];

        int ans = helper(n - 1, amount, coins);

        return ans == (int)1e9 ? -1 : ans;
    }

    private int helper(int idx, int amount, int[] coins) {
        // Base case: check if amount can be formed using first coin
        if (idx == 0) {
            if (amount % coins[0] == 0) return amount / coins[0];
            return (int)1e9;
        }

        if (dp[idx][amount] != null) return dp[idx][amount];

        int notPick = helper(idx - 1, amount, coins);

        int pick = (int)1e9;
        if (coins[idx] <= amount) {
            pick = 1 + helper(idx, amount - coins[idx], coins);
        }

        return dp[idx][amount] = Math.min(pick, notPick);
    }
}

        dp = new Integer[n + 1];
        return helper(n); // 0 se n tak jaane ke total ways
    }

    private int helper(int idx) { // 0 se idx tak jaane ke total ways
        if (idx == 0) return 1; // Reached top exactly
        if (idx < 0) return 0;  // Invalid step (overshoot)

        if (dp[idx] != null) return dp[idx];

        int oneStep = helper(idx - 1); // 1 step climb
        int twoStep = helper(idx - 2); // 2 steps climb

        return dp[idx] = oneStep + twoStep;
    }

    private boolean canBreak(int start, String s, Set<String> dict) {
        if (start == s.length()) return true; // Reached end → valid segmentation

        if (dp[start] != null) return dp[start]; // Return memoized result

        // Try every possible prefix starting from 'start'
        for (int end = start + 1; end <= s.length(); end++) {
            String prefix = s.substring(start, end);
            // If prefix is valid word and rest of string can be segmented
            if (dict.contains(prefix) && canBreak(end, s, dict)) {
                return dp[start] = true; // Memoize and return true
            }
        }

        return dp[start] = false; // No valid segmentation found
    }

  • Edit Distance
    class Solution { //DSAQuickRevision
    public int editDistance(String word1, String word2) {
        int n = word1.length();
        int m = word2.length();
        int[][] dp = new int[n][m];
        for (int[] row : dp) {
            Arrays.fill(row, -1); // Initialize DP table with -1
        }
        return helper(n - 1, m - 1, word1, word2, dp);
    }

    private int helper(int i, int j, String word1, String word2, int[][] dp) {
        if (i < 0) return j + 1; // Insert all remaining characters from word2
        if (j < 0) return i + 1; // Delete all remaining characters from word1

        if (dp[i][j] != -1) return dp[i][j]; // Return precomputed value

        if (word1.charAt(i) == word2.charAt(j)) {
            return dp[i][j] = helper(i - 1, j - 1, word1, word2, dp); // No operation needed
        }

        // Replace: Change word1[i] to word2[j] and move diagonally
        int replace = helper(i - 1, j - 1, word1, word2, dp);

        // Delete: Remove word1[i] and move up in word1
        int delete = helper(i - 1, j, word1, word2, dp);

        // Insert: Add word2[j] to word1 and move left in word2
        int insert = helper(i, j - 1, word1, word2, dp);

        // Take the minimum of all three operations and add 1 for the current operation
        return dp[i][j] = 1 + Math.min(replace, Math.min(delete, insert));
    }
}

    class Solution { //DSAQuickRevision
    Integer[][] dp; // Memoization table

    public int LIS(int[] nums) {
        int n = nums.length;
        dp = new Integer[n][n + 1]; // DP table: [index][prevIndex + 1]
        return helper(0, -1, nums); // Start with index 0 and no previous index
    }

    private int helper(int idx, int prevIdx, int[] nums) {
        if (idx == nums.length) return 0; // Base case: no more elements

        if (dp[idx][prevIdx + 1] != null) return dp[idx][prevIdx + 1]; // Return cached result

        // Option 1: Skip current element
        int notPick = helper(idx + 1, prevIdx, nums);

        // Option 2: Include current element if valid
        int pick = 0;
        if (prevIdx == -1 || nums[idx] > nums[prevIdx]) {
            pick = 1 + helper(idx + 1, idx, nums);
        }

        // Cache and return the maximum of the two choices
        return dp[idx][prevIdx + 1] = Math.max(notPick, pick);
    }
}

                int pick = 0;
                if (prevIdx == -1 || nums[idx] > nums[prevIdx]) {
                    pick = 1 + dp[idx + 1][idx + 1];
                }

                dp[idx][prevIdx + 1] = Math.max(pick, notPick);
            }
        }

        // Final answer is for index 0 and no previous index (prevIdx = -1 → +1 = 0)
        return dp[0][0];
    }

    public boolean isMatch(String s, String p) {
        return match(0, 0, s, p);
    }

    private boolean match(int i, int j, String s, String p) {
        String key = i + "," + j;
        if (memo.containsKey(key)) return memo.get(key);

        // Base case: pattern khatam ho gaya
        if (j == p.length()) {
            return i == s.length(); // string bhi khatam ho?
        }

        // First character match ho raha?
        boolean firstMatch = (i < s.length()) &&
                             (s.charAt(i) == p.charAt(j) || p.charAt(j) == '.');

        // Check for '*'
        if (j + 1 < p.length() && p.charAt(j + 1) == '*') {
            // 2 options: 
            // 1. Ignore x* → move j by 2
            // 2. Consume if first matches → move i
            boolean ans = match(i, j + 2, s, p) || (firstMatch && match(i + 1, j, s, p));
            memo.put(key, ans);
            return ans;
        } else {
            // Move both forward if first char matched
            boolean ans = firstMatch && match(i + 1, j + 1, s, p);
            memo.put(key, ans);
            return ans;
        }
    }

    class Solution {
    private int[][] memo;
    private char[][] grid;
    private int m, n, maxSide;

    public int maximalSquare(char[][] matrix) {
        grid = matrix;
        m = matrix.length;
        n = matrix[0].length;
        memo = new int[m][n];
        for (int[] row : memo) Arrays.fill(row, -1); // init memo

        maxSide = 0;

        // Try to start square from every cell
        for (int i = 0; i < m; i++) {
            for (int j = 0; j < n; j++) {
                maxSide = Math.max(maxSide, dfs(i, j));
            }
        }

        return maxSide * maxSide; // return area
    }

    // Returns largest square side starting at (i, j)
    private int dfs(int i, int j) {
        if (i >= m || j >= n) return 0;          // out of bounds
        if (memo[i][j] != -1) return memo[i][j]; // already solved
        if (grid[i][j] == '0') return memo[i][j] = 0; // can't form square

        // check right, down, diagonal
        int right = dfs(i, j + 1);
        int down = dfs(i + 1, j);
        int diag = dfs(i + 1, j + 1);

        return memo[i][j] = 1 + Math.min(right, Math.min(down, diag));
    }
}

    class Solution { //DSAConceptRevision
    public boolean wildCard(String s, String p) {
        int n = s.length();  // Length of the input string
        int m = p.length();  // Length of the pattern

        // DP table for memoization (initialized with null, meaning uncomputed states)
        Boolean[][] dp = new Boolean[n][m];
        for (Boolean[] row : dp) Arrays.fill(row, null);  // Filling the DP table with null initially

        // Recursive function with memoization (starting from n-1, m-1)
        return match(n - 1, m - 1, s, p, dp);  // Start the matching from the end of both strings
    }

    private boolean match(int i, int j, String s, String p, Boolean[][] dp) {
        // Base case: If both strings are fully matched (both are exhausted)
        if (i < 0 && j < 0) return true;

        // If the pattern is exhausted but the string is not, it's an automatic mismatch
        if (j < 0) return false;

        // If the string is exhausted, the pattern should only contain '*' to still match
        if (i < 0) {
            // All remaining pattern characters must be '*' to match the empty string
            for (int k = j; k >= 0; k--) {
                if (p.charAt(k) != '*') return false;  // If any non '*' character is found, return false
            }
            return true;  // All characters in the pattern are '*' so it matches
        }

        // If this state (i, j) is already computed, simply return the stored result from the DP table
        if (dp[i][j] != null) return dp[i][j];  // Memoization check

        // Variable to store the result for this state (i, j)
        Boolean ans = false;

        // Case 1: Direct character match OR '?' wildcard (which matches any single character)
        if (s.charAt(i) == p.charAt(j) || p.charAt(j) == '?') {
            // If characters match or pattern[j] is '?', move both string and pattern indexes
            ans = match(i - 1, j - 1, s, p, dp);  // Recursively match the next characters
        } 
        // Case 2: '*' wildcard (can match empty sequence or any sequence of characters)
        else if (p.charAt(j) == '*') {
            // '*' can match an empty sequence (move the pattern forward)
            // OR it can consume one character from the string and keep '*' in play (move the string forward)
            ans = match(i, j - 1, s, p, dp) || match(i - 1, j, s, p, dp);
        }

        // Store the result in the DP table for this state (i, j)
        dp[i][j] = ans;  // Memoization: store the computed result for future reference

        return ans;  // Return the final result for this state
    }
}

    class Solution { //DSAQuickRevision
    int[][] dp;

    public int lcs(String s1, String s2) {
        int len1 = s1.length();
        int len2 = s2.length();
        dp = new int[len1][len2]; // DP table for memoization
        for (int[] row : dp) {
            Arrays.fill(row, -1); // Initialize with -1 to indicate uncomputed states
        }

        return helper(len1 - 1, len2 - 1, s1, s2); // Start recursion from the last indices
    }

    public int helper(int ind1, int ind2, String s1, String s2) {
        if (ind1 < 0 || ind2 < 0) return 0; // Base case: If either string is exhausted, LCS is 0

        if (dp[ind1][ind2] != -1) return dp[ind1][ind2]; // Return precomputed value if available

        // If characters match, add 1 to the LCS count and move diagonally in the DP table
        if (s1.charAt(ind1) == s2.charAt(ind2)) {
            return dp[ind1][ind2] = 1 + helper(ind1 - 1, ind2 - 1, s1, s2);
        }

        // If characters don't match, take the max LCS by moving one step in either string
        return dp[ind1][ind2] = Math.max(
            helper(ind1 - 1, ind2, s1, s2), 
            helper(ind1, ind2 - 1, s1, s2)
        );
    }

    int[][] dp;

    public int longestCommonSubsequence(String s1, String s2) {
        int len1 = s1.length();
        int len2 = s2.length();
        dp = new int[len1][len2];
        for (int[] row : dp) {
            Arrays.fill(row, -1);
        }
        return helper(len1 - 1, len2 - 1, s1, s2);
    }

    public int helper(int ind1, int ind2, String s1, String s2) {
        if (ind1 < 0 || ind2 < 0) return 0; // Base case: If any string is exhausted
        if (dp[ind1][ind2] != -1) return dp[ind1][ind2]; // Use cached result if available

        // If characters match, move diagonally
        if (s1.charAt(ind1) == s2.charAt(ind2)) {
            return dp[ind1][ind2] = 1 + helper(ind1 - 1, ind2 - 1, s1, s2);
        }

        // If characters don't match, take the maximum of the two possibilities
        return dp[ind1][ind2] = Math.max(
            helper(ind1 - 1, ind2, s1, s2), 
            helper(ind1, ind2 - 1, s1, s2)
        );
    }
}

private int lcs(String s1, String s2) {
    int n = s1.length();
    int m = s2.length();

    int[][] dp = new int[n + 1][m + 1];

    // Initialize the first row and first column to 0
    for (int i = 0; i <= n; i++) {
        dp[i][0] = 0;
    }
    for (int i = 0; i <= m; i++) {
        dp[0][i] = 0;
    }

    for (int ind1 = 1; ind1 <= n; ind1++) {
        for (int ind2 = 1; ind2 <= m; ind2++) {
            if (s1.charAt(ind1 - 1) == s2.charAt(ind2 - 1))
                dp[ind1][ind2] = 1 + dp[ind1 - 1][ind2 - 1];
            else
                dp[ind1][ind2] = Math.max(dp[ind1 - 1][ind2], dp[ind1][ind2 - 1]);
        }
    }
    // Return the result
    return dp[n][m];
}
