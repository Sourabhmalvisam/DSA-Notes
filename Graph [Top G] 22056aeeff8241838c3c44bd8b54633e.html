    class Solution { //DSAConceptRevision
    // Function to find the number of islands.
    public int numIslands(char[][] grid) {
        int rows = grid.length;
        int cols = grid[0].length;
        boolean[][] visited = new boolean[rows][cols];
        int islandCount = 0;

        // Traverse the grid
        for (int i = 0; i < rows; i++) {
            for (int j = 0; j < cols; j++) {
                // Check if the cell is unvisited and represents land
                if (!visited[i][j] && grid[i][j] == '1') {
                    // Explore the island using BFS
                    bfs(grid, visited, i, j);
                    // Increment the count as a new island is found
                    islandCount++;
                }
            }
        }
        return islandCount;
    }

    static class Pair {
        int row;
        int col;

        Pair(int r, int c) {
            row = r;
            col = c;
        }
    }

    // Helper function for BFS traversal
    public static void bfs(char[][] grid, boolean visited[][], int startRow, int startCol) {
        int rows = grid.length;
        int cols = grid[0].length;
        Queue<Pair> queue = new LinkedList<>();
        queue.add(new Pair(startRow, startCol));
        visited[startRow][startCol] = true;

        while (!queue.isEmpty()) {
            Pair current = queue.poll();
            int currentRow = current.row;
            int currentCol = current.col;

            // Explore all 8 directions
            for (int i = -1; i <= 1; i++) {
                for (int j = -1; j <= 1; j++) {
                    int newRow = currentRow + i;
                    int newCol = currentCol + j;

                    // Check if the new position is within bounds and represents unvisited land
                    if (newRow >= 0 && newCol >= 0 && newRow < rows && newCol < cols && !visited[newRow][newCol]
                            && grid[newRow][newCol] == '1') {
                        // Mark as visited and add to the queue for further exploration
                        visited[newRow][newCol] = true;
                        queue.add(new Pair(newRow, newCol));
                    }
                }
            }
        }
    }

                } else if (grid[i][j] == 2) {

                }
            }

        while (!queue.isEmpty()) {
            Orange currentOrange = queue.poll();
            int currentRow = currentOrange.row;
            int currentCol = currentOrange.col;
            int currentTime = currentOrange.time;
            maxTime = Math.max(currentTime, maxTime);

            // Explore neighboring oranges
            for (int[] dir : directions) {
                int newRow = currentRow + dir[0];
                int newCol = currentCol + dir[1];

                // Check if the neighbor is within bounds and is a fresh orange
                if (newRow >= 0 && newCol >= 0 && newRow < rows && newCol < cols &&
                    visited[newRow][newCol] == 0 && grid[newRow][newCol] == 1) {
                    visited[newRow][newCol] = 2;

                    rottenOrangesCount++; // Imp: why we whote this here only and not inside while?
                }
            }
        }

        if (rottenOrangesCount == freshOrangesCount) {

        return -1;

	public static String findOrder(String[] dict, int N, int K) {
		// Create an adjacency list to represent the directed graph of characters.
		List <List<Integer>> adj = new ArrayList<>();

		for (int i = 0; i < K; i++) {
			adj.add(new ArrayList<>());
		}

		// Populate the adjacency list based on the given dictionary of words.
		for (int i = 0; i < N - 1; i++) {
			String word1 = dict[i];
			String word2 = dict[i + 1];
			int len = Math.min(word1.length(), word2.length());

			for (int ptr = 0; ptr < len; ptr++) {
				if (word1.charAt(ptr) != word2.charAt(ptr)) {
					adj.get(word1.charAt(ptr) - 'a').add(word2.charAt(ptr) - 'a');
					// Break once the first differing character is found.
					break; // ye matt bhoolna
				}
			}
		}

		// Perform topological sort to get the order of characters.
		List<Integer> topo = topoSort(K, adj);

		// Convert the topological order to a string.
		StringBuilder result = new StringBuilder();

		for (int it: topo) {
			result.append((char)(it + 'a'));
		}

		return result.toString();
	}

	// Helper function for topological sort.
	private static List<Integer> topoSort(int V, List <List<Integer>> adj) {
		int indegree[] = new int[V];

		for (int i = 0; i < V; i++) {
			for (int it: adj.get(i)) {
				indegree[it]++;
			}
		}

		Queue<Integer> q = new LinkedList<>();

		for (int i = 0; i < V; i++) {
			if (indegree[i] == 0) {
				q.add(i);
			}
		}

		List<Integer> topo = new ArrayList<>();

		while (!q.isEmpty()) {
			int node = q.poll();
			topo.add(node);

			for (int it: adj.get(node)) {
				indegree[it]--;

				if (indegree[it] == 0)
					q.add(it);
			}
		}

		return topo;
	}

    class Solution { //DSAConceptRevision
    int[] par;
    int[] size;

    // Initialize Union-Find structure
    void init(int n) {
        par = new int[n];
        size = new int[n];
        for (int i = 0; i < n; i++) {
            par[i] = i;
            size[i] = 1;
        }
    }

    // Find with path compression
    int find(int x) {
        if (x == par[x])
            return x;
        return par[x] = find(par[x]);
    }

    // Union by size
    void union(int a, int b) {
        int parA = find(a);
        int parB = find(b);
        if (parA == parB)
            return;

        if (size[parA] < size[parB]) {
            par[parA] = parB;
            size[parB] += size[parA];
        } else {
            par[parB] = parA;
            size[parA] += size[parB];
        }
    }

    // Check if cell is within grid boundaries
    boolean isValid(int r, int c, int n) {
        return r >= 0 && c >= 0 && r < n && c < n;
    }

    int[][] dirs = { { 1, 0 }, { -1, 0 }, { 0, 1 }, { 0, -1 } };

    public int largestIsland(int[][] grid) {
        int n = grid.length;
        init(n * n);

        // Union-Find preprocessing
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < n; j++) {
                if (grid[i][j] == 0)
                    continue;
                for (int[] dir : dirs) {
                    int nr = dir[0] + i;
                    int nc = dir[1] + j;
                    if (isValid(nr, nc, n) && grid[nr][nc] == 1) {
                        int node = i * n + j;
                        int neiNode = nr * n + nc;
                        union(node, neiNode);
                    }
                }
            }
        }

        // Trying to convert each 0 to 1 and see the size of the connected component
        int max = 0;
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < n; j++) {
                if (grid[i][j] == 1)
                    continue;
                Set<Integer> components = new HashSet<>();

                for (int[] dir : dirs) {
                    int nr = dir[0] + i;
                    int nc = dir[1] + j;
                    if (isValid(nr, nc, n) && grid[nr][nc] == 1) {
                        components.add(find(nr * n + nc)); // Add only top parent
                    }
                }
                int totalSize = 0;
                for (int node : components) {
                    totalSize+= size[node];
                }
                max = Math.max(max, totalSize+ 1); // +1 for the converted cell
            }
        }

        // If all cells are 1, or max is 0 (no 0s were converted to 1s)
        for (int cellNo = 0; cellNo < n * n; cellNo++) {
            max = Math.max(max, size[find(cellNo)]);
        }

        return max;
    }
}

    class Solution { //DSAQuickRevision
    public int CheapestFlight(int n, int[][] edges, int src, int dst, int k) {
        // Adjacency list to represent the graph
        ArrayList<ArrayList<Edge>> adjacencyList = new ArrayList<>();
        for (int i = 0; i < n; i++) {
            adjacencyList.add(new ArrayList<>());
        }

        // Adding edges to the adjacency list
        for (int[] edge : edges) {
            adjacencyList.get(edge[0]).add(new Edge(edge[1], edge[2]));
        }

        // Array to store minimum distances from source to each node
        int[] minDistances = new int[n];
        Arrays.fill(minDistances, Integer.MAX_VALUE);
        minDistances[src] = 0;

        // Queue for BFS traversal
        Queue<Triple> queue = new LinkedList<>();
        queue.offer(new Triple(0, src, 0));

        // BFS traversal
        while (!queue.isEmpty()) {
            Triple front = queue.poll();

            // Check if steps exceed the limit
            if (front.steps > k) {
                continue;
            }

            // Explore neighbors and update minimum distances
            for (Edge neighbor : adjacencyList.get(front.node)) {
                int totalDistance = front.distance + neighbor.weight;

                // Update if the new distance is smaller
                if (totalDistance < minDistances[neighbor.destination]) {
                    minDistances[neighbor.destination] = totalDistance;
                    queue.offer(new Triple(front.steps + 1, neighbor.destination, totalDistance));
                }
            }
        }

        // If destination is not reachable, return -1; otherwise, return the minimum
        // distance
        return minDistances[dst] == Integer.MAX_VALUE ? -1 : minDistances[dst];
    }

    // Pair class to represent an edge in the graph
    class Edge {
        int destination;
        int weight;

        public Edge(int destination, int weight) {
            this.destination = destination;
            this.weight = weight;
        }
    }

    // Triple class to represent a step, node, and distance during BFS
    class Triple {
        int steps;
        int node;
        int distance;

        public Triple(int steps, int node, int distance) {
            this.steps = steps;
            this.node = node;
            this.distance = distance;
        }
    }

}

    public int countDistinctIslands(int[][] grid) {
        int rows = grid.length, cols = grid[0].length;
        Set<Set<String>> uniqueIslands = new HashSet<>();
        boolean[][] visited = new boolean[rows][cols];

        for (int i = 0; i < rows; i++) {
            for (int j = 0; j < cols; j++) {
                if (grid[i][j] == 1 && !visited[i][j]) {
                    Set<String> shape = new HashSet<>();
                    dfs(grid, visited, i, j, i, j, shape);
                    uniqueIslands.add(shape);
                }
            }
        }
        return uniqueIslands.size();
    }

        if (i < 0 || j < 0 || i >= grid.length || j >= grid[0].length || grid[i][j] == 0 || visited[i][j]) {
            return;
        }

        visited[i][j] = true;
        shape.add((i - baseX) + "," + (j - baseY)); // Store relative position

        // Explore all 4 directions
        dfs(grid, visited, i + 1, j, baseX, baseY, shape); // Down
        dfs(grid, visited, i - 1, j, baseX, baseY, shape); // Up
        dfs(grid, visited, i, j + 1, baseX, baseY, shape); // Right
        dfs(grid, visited, i, j - 1, baseX, baseY, shape); // Left
    }

    class Solution { //DSAConceptRevision

	public List < List<String>> accountsMerge(List < List<String>> accounts) {
		int n = accounts.size();

		// Initialize Disjoint Set
		DisjointSet ds = new DisjointSet(n);

		// Map to store the association between emails and their corresponding nodes
		HashMap<String, Integer> mapMailNode = new HashMap<String, Integer>();

		// Step 1: Build associations and perform unions
		for (int i = 0; i < n; i++) {
			for (int j = 1; j < accounts.get(i).size(); j++) {
				String mail = accounts.get(i).get(j);

				// If the email is encountered for the first time
				if (!mapMailNode.containsKey(mail)) {
					mapMailNode.put(mail, i);
				} else {
					// Union the current node with the node associated with the existing email
					ds.union(i, mapMailNode.get(mail));
				}
			}
		}

		// Step 2: Build mergedMail array to group emails based on their root nodes
		ArrayList<String> [] mergedMail = new ArrayList[n];

		for (int i = 0; i < n; i++)
			mergedMail[i] = new ArrayList<String>();

		for (Map.Entry<String, Integer> it: mapMailNode.entrySet()) {
			String mail = it.getKey();
			int node = ds.find(it.getValue()); // Top G: Important Step hai ye.
			mergedMail[node].add(mail);
		}

		// Step 3: Construct the final result
		List <List<String>> ans = new ArrayList<>();

		for (int i = 0; i < n; i++) {
			if (mergedMail[i].size() == 0)
				continue;

			// Sort the emails within each group for consistency
			Collections.sort(mergedMail[i]);

			// Build the merged account information
			List<String> temp = new ArrayList<>();
			temp.add(accounts.get(i).get(0)); // Name (first element in each account)
			for (String it: mergedMail[i]) {
				temp.add(it); // Append sorted emails
			}

			ans.add(temp); // Add the merged account to the result
		}

		return ans;
	}

	class DisjointSet {
		// Array to store parent nodes
		int par[];

		// Array to store rank for rank-based optimization
		int rank[];

		// Initialization: Set each element as a parent of itself and rank as 0
		public DisjointSet(int n) {
			par = new int[n];
			rank = new int[n];

			for (int i = 0; i < n; i++) {
				par[i] = i;
				rank[i] = 0;
			}
		}

		// Find operation with path compression
		public int find(int x) {
			if (x == par[x])
				return x;
			return par[x] = find(par[x]); // Path compression: Update parent to root during find
		}

		// Union operation with rank-based optimization
		public void union(int a, int b) {
			int parA = find(a);
			int parB = find(b);
			
			if(parA == parB) return; // Important.
			
			if (rank[parA] == rank[parB]) {
				par[parA] = parB;
				rank[parB] += 1;
			} else if (rank[parA]<rank[parB]) {
				par[parA] = parB;
			} else {
				par[parB] = parA;

			}
		}
	}

            return;

        dfs(grid, sr + 1, sc, original, newColor); // Down
        dfs(grid, sr - 1, sc, original, newColor); // Up
        dfs(grid, sr, sc - 1, original, newColor); // Left
        dfs(grid, sr, sc + 1, original, newColor); // Right
    }

    public int shortestPath(int[][] grid, int[] source,

            source[1] == destination[1])
            return 0;

        /* Queue data structure to store the pairs of the

        int n = grid.length;

        // Distance matrix
        int[][] dist = new int[n][m];
        for (int[] row : dist) {
            Arrays.fill(row, Integer.MAX_VALUE);
        }
        
        // Distance of source from itself is zero
        dist[source[0]][source[1]] = 0;
        
        // Add the source to queue
        q.add(new int[]{0, source[0], source[1]});

        // Until the queue is empty
        while (!q.isEmpty()) {
            // Get the element
            int[] it = q.poll();
            
            int dis = it[0]; // Distance
            int row = it[1]; // Row of cell
            int col = it[2]; // Column of cell

            // Iterate through all the neighbors
            for (int i = 0; i < 4; i++) {
                
                // Coordinates of the new cell
                int newRow = row + delRow[i];
                int newCol = col + delCol[i];

                /* Checking the validity of the cell and 
                updating if a shorter distance is found */
                if (isValid(newRow, newCol, n, m) && 
                    grid[newRow][newCol] == 1 && 
                    dis + 1 < dist[newRow][newCol]) {

        int[][] maxDiff = new int[n][m];

        {diff, {row of cell, column of cell}} */
        PriorityQueue<int[]> pq = new PriorityQueue<>(Comparator.comparingInt(a -> a[0]));

        // Mark the initial position as 0
        maxDiff[0][0] = 0;

        // Push the starting cell to min-heap
        pq.add(new int[]{0, 0, 0});

        // Until the min-heap is not empty
        while (!pq.isEmpty()) {

            /* Get the cell with minimum 
            difference (effort) */
            int[] p = pq.poll();

            // Get the difference
            int diff = p[0];

            int row = p[1]; // Row
            int col = p[2]; // Column

            /* If the destination cell is reached, 
            return the difference */
            if (row == n - 1 && col == m - 1)
                return diff;

            // Traverse its neighbors
            for (int i = 0; i < 4; i++) {

                /* Get the coordinates 
                of neighboring cell */
                int newRow = row + delRow[i];
                int newCol = col + delCol[i];

                // Check if the cell is valid
                if (isValid(newRow, newCol, n, m)) {

                    /* Get the current difference 
                    in heights of cells */
                    int currDiff = Math.abs(heights.get(newRow).get(newCol) - heights.get(row).get(col));

                    /* Check if the current difference is 
                    less than earlier known difference */
                    if (Math.max(currDiff, diff) < maxDiff[newRow][newCol]) {

                        // Store the current difference
                        maxDiff[newRow][newCol] = Math.max(currDiff, diff);

                        // Add the new pair found in the min-heap
                        pq.add(new int[]{Math.max(currDiff, diff), newRow, newCol});
                    }
                }
            }
        }

    static int rank[];

    // Initialize Union-Find structures
    static void init(int N) {
        par = new int[N];
        rank = new int[N];
        for (int i = 0; i < N; i++) {
            par[i] = i;
        }
    }

    // Find with path compression
    static int find(int x) {
        if (x == par[x]) return x;
        return par[x] = find(par[x]);
    }

    // Union by rank
    static void union(int a, int b) {
        int parA = find(a);
        int parB = find(b);

        if (parA == parB) return; // Important.

        if (rank[parA] == rank[parB]) {
            par[parA] = parB;
            rank[parB]++;
        } else if (rank[parA] < rank[parB]) {
            par[parA] = parB;
        } else {
            par[parB] = parA;
        }
    }

        List<Edge> edges = new ArrayList<>();

        for (int i = 0; i < V; i++) {
            for (List<Integer> edge : adj.get(i)) {
                int neighbor = edge.get(0);
                int weight = edge.get(1);
                edges.add(new Edge(i, neighbor, weight));
                
            }
        }

        int count = 0;
        int mstCost = 0;

        for (int i = 0; i < edges.size(); i++) {

            int u = e.src;

            int weight = e.weight;

    public int numberOfEnclaves(int[][] grid) {
        int rows = grid.length;
        int cols = grid[0].length;

        // Step 1: Mark all land cells connected to the boundary
        for (int i = 0; i < rows; i++) {
            if (grid[i][0] == 1) dfs(grid, i, 0, rows, cols); // Left boundary
            if (grid[i][cols - 1] == 1) dfs(grid, i, cols - 1, rows, cols); // Right boundary
        }
        for (int j = 0; j < cols; j++) {
            if (grid[0][j] == 1) dfs(grid, 0, j, rows, cols); // Top boundary
            if (grid[rows - 1][j] == 1) dfs(grid, rows - 1, j, rows, cols); // Bottom boundary
        }

        // Step 2: Count remaining land cells
        int enclaveCount = 0;
        for (int i = 0; i < rows; i++) {
            for (int j = 0; j < cols; j++) {
                if (grid[i][j] == 1) {
                    enclaveCount++;
                }
            }
        }

        return enclaveCount;
    }

    // Helper function to perform DFS and mark boundary-connected land cells as visited
    private void dfs(int[][] grid, int r, int c, int rows, int cols) {
        if (r < 0 || c < 0 || r >= rows || c >= cols || grid[r][c] == 0) return;

        grid[r][c] = 0; // Mark the cell as visited by converting it to sea
        dfs(grid, r + 1, c, rows, cols); // Down
        dfs(grid, r - 1, c, rows, cols); // Up
        dfs(grid, r, c + 1, rows, cols); // Right
        dfs(grid, r, c - 1, rows, cols); // Left
    }

    class Solution { //DSAStriver
    /* Function to find the shortest 
    path from node 1 to node n */
    public List<Integer> shortestPath(int n, int m, int[][] edges) {
        
        // Adjacency list to store graph
        List<List<int[]>> adj = new ArrayList<>();
        for (int i = 0; i <= n; i++) {
            adj.add(new ArrayList<>());
        }
        
        // Adding the edges to the graph
        for (int[] edge : edges) {
            adj.get(edge[0]).add(new int[]{edge[1], edge[2]});
            adj.get(edge[1]).add(new int[]{edge[0], edge[2]});
        }
        
        /* Using priority queue to 
        implement Dijkstra Algorithm */
        PriorityQueue<int[]> pq = new PriorityQueue<>(Comparator.comparingInt(a -> a[0]));
        
        // Distance array
        int[] dist = new int[n + 1];
        Arrays.fill(dist, Integer.MAX_VALUE);
        
        // Parent array
        int[] parent = new int[n + 1];
        
        /* Marking each node as 
        its own parent initially */
        for (int i = 1; i <= n; i++) {
            parent[i] = i;
        }
        
        /* Distance of source node 
        (node 1) to itself is zero */
        dist[1] = 0;
        
        // Push the source node to the queue.
        pq.add(new int[]{0, 1});
        
        // Until the queue is empty
        while (!pq.isEmpty()) {
            /* Get the pair containing node having 
            minimum distance from source node */
            int[] curr = pq.poll();
            
            int node = curr[1]; // node
            int dis = curr[0]; // distance
            
            // Iterate through the neighbors
            for (int[] neighbor : adj.get(node)) {
                int adjNode = neighbor[0]; // node
                int edWt = neighbor[1]; // edge weight
                
                /* If the tentative distance to 
                reach adjacent node is smaller 
                than the known distance */
                if (dis + edWt < dist[adjNode]) {
                    
                    // Update the known distance
                    dist[adjNode] = dis + edWt;
                    
                    // Push the new pair to priority queue
                    pq.add(new int[]{dis + edWt, adjNode});
                    
                    /* Update the parent of the adjNode 
                    to the recent node(where it came from) */
                    parent[adjNode] = node;
                }
            }
        }
        
        /* If distance to the node could not be found, 
        return an array containing -1. */
        if (dist[n] == Integer.MAX_VALUE) {
            return Arrays.asList(-1);
        }
        
        // Array to store the path
        List<Integer> path = new ArrayList<>();
        
        // Start from the destination node
        int node = n;
        
        /* Iterate backwards from destination 
        to source through the parent array */
        while (parent[node] != node) {
            
            // Add the node to the path
            path.add(node); 
            
            // Take a step back
            node = parent[node];
        }
        
        // Add the source node to the path
        path.add(1);
        
        /* Since the path stored is in a 
        reverse order, reverse the array 
        to get the actual path */
        Collections.reverse(path);
        
        // Add the path weight in the beginning
        path.add(0, dist[n]);
        
        // Return the result
        return path;
    }
}

    sizes of different set of vertices */
    int[] rank, parent, size;

    // Constructor
    DisjointSet(int n) {
        rank = new int[n + 1];
        parent = new int[n + 1];
        size = new int[n + 1];
        for (int i = 0; i <= n; i++) {
            parent[i] = i;
            size[i] = 1;
        }
    }

    // Function to find ultimate parent
    int findUPar(int node) {
        if (node == parent[node])
            return node;
        return parent[node] = findUPar(parent[node]);
    }

    // Function to implement union by size
    void unionBySize(int u, int v) {
        int ulp_u = findUPar(u);
        int ulp_v = findUPar(v);
        if (ulp_u == ulp_v) return;
        if (size[ulp_u] < size[ulp_v]) {
            parent[ulp_u] = ulp_v;
            size[ulp_v] += size[ulp_u];
        }
        else {
            parent[ulp_v] = ulp_u;
            size[ulp_u] += size[ulp_v];
        }
    }
}
