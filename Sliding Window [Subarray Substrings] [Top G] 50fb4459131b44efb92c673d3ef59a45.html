    class Solution { //DSAQuickRevision
    public int longestNonRepeatingSubstring(String s) {
        // Use a HashMap to store character frequencies in the current window
        HashMap<Character, Integer> map = new HashMap<>();
        int maxLength = 0; // To store the maximum length of the substring
        int i = 0; // Start of the sliding window

        // Traverse the string
        for (int j = 0; j < s.length(); j++) {
            // Step 1: Add the current character to the map and update its frequency
            char ch = s.charAt(j);
            map.put(ch, map.getOrDefault(ch, 0) + 1);

            // Step 2: If a character repeats (frequency > 1), shrink the window
            while (map.get(ch) > 1) { // Top G Note: While loop pehle likhna, XD. (explain why?)
                char charAtStart = s.charAt(i);
                map.put(charAtStart, map.get(charAtStart) - 1);

                // If the frequency of a character becomes 0, remove it from the map
                if (map.get(charAtStart) == 0) {
                    map.remove(charAtStart);
                }

                i++; // Move the start of the window forward
            }

            // Step 3: Update the maximum length of the substring
			          maxLength = Math.max(maxLength, j - i + 1);
        }

        return maxLength; // Return the maximum length of substring without repeating characters
    }
}
  • Sliding Window Maximum (Deque pattern, interview favorite)
    class Solution { //DSAConceptRevision
    public int[] maxSlidingWindow(int[] nums, int k) {
        LinkedList<Integer> deque = new LinkedList<>();
        ArrayList<Integer> result = new ArrayList<>();

        int i = 0;
        for (int j = 0; j < nums.length; j++) {
            // Step 1: Remove all elements smaller than the current element from the deque
            while (!deque.isEmpty() && deque.getLast() < nums[j]) {
                deque.removeLast();
            }
            deque.addLast(nums[j]); // Add the current value to the deque

            // Step 2: Check if the window size has reached k
            if (j - i + 1 == k) {
                result.add(deque.getFirst()); // Add the maximum in the current window to the result
                if (deque.getFirst() == nums[i]) { // Step 3: Remove the element at the start of the window
                    deque.removeFirst();
                }
                i++; // Step 4: Slide the window
            }
        }

        // Convert the result ArrayList to an array
        int[] res = new int[result.size()];
        for (int idx = 0; idx < result.size(); idx++) {
            res[idx] = result.get(idx);
        }
        return res;
    }
}
  • Minimum Window Substring
    class Solution { //DSAConceptRevision
    public String minWindow(String s, String t) {
        // Edge case: If s or t is null, or s is shorter than t
        if (s == null || t == null || s.length() < t.length()) {
            return "";
        }

        // Frequency map for characters in t
        Map<Character, Integer> tMap = new HashMap<>();
        for (char c : t.toCharArray()) {
            tMap.put(c, tMap.getOrDefault(c, 0) + 1);
        }

        // Sliding window
        Map<Character, Integer> windowMap = new HashMap<>();
        int l = 0; // Left pointer of the window
        int matched = 0; // Count of characters that match the frequency in tMap
        int minLen = Integer.MAX_VALUE; // Minimum window length
        int start = 0; // Start index of the minimum window

        // Iterate over the string with the right pointer (r)
        for (int r = 0; r < s.length(); r++) {
            char rightChar = s.charAt(r);
            windowMap.put(rightChar, windowMap.getOrDefault(rightChar, 0) + 1);

            // Check if the current character satisfies the frequency in tMap
            if (tMap.containsKey(rightChar) && windowMap.get(rightChar) == tMap.get(rightChar)) {
                matched++;
            }

            // When all characters are matched, try to shrink the window
            while (matched == tMap.size()) {
                // Update the minimum window if needed
                if (r - l + 1 < minLen) {
                    minLen = r - l + 1;
                    start = l;
                }

                char leftChar = s.charAt(l);
                windowMap.put(leftChar, windowMap.get(leftChar) - 1);

                // Check if the character no longer satisfies the frequency
                if (tMap.containsKey(leftChar) && windowMap.get(leftChar) < tMap.get(leftChar)) {
                    matched--;
                }

                l++; // Shrink the window from the left
            }
        }

        // Return the minimum window substring
        return minLen == Integer.MAX_VALUE ? "" : s.substring(start, start + minLen);
    }
}
  • Longest Repeating Character Replacement (Replace at most k chars to make longest repeating substring)
    class Solution { //DSAConceptRevision
    public int characterReplacement(String s, int k) {
        int i = 0; // Window start index
        int maxFreq = 0; // Most frequent character count in current window
        int maxLength = 0; // Final answer
        HashMap<Character, Integer> map = new HashMap<>(); // Character frequency map for current window

        for (int j = 0; j < s.length(); j++) {
            char currChar = s.charAt(j);
            
            // 1. Expand the window by including s[j]
            map.put(currChar, map.getOrDefault(currChar, 0) + 1);
            
            // Update the max frequency seen so far in the window
            maxFreq = Math.max(maxFreq, map.get(currChar));

            // 2. Check if window is invalid
            // Condition: total letters to change = (window size - maxFreq) > k
            while ((j - i + 1) - maxFreq > k) {
                // Shrink window from left if too many changes needed
                char leftChar = s.charAt(i);
                map.put(leftChar, map.get(leftChar) - 1);
                i++; // Move window forward
            }

            // 3. After ensuring window is valid, update max length
	        if((j - i + 1) - maxFreq <= k) maxLength = Math.max(maxLength, j - i + 1);
        }

        return maxLength;
    }
}
  • Subarray Product Less Than K
    class Solution {
    public int numSubarrayProductLessThanK(int[] nums, int k) {
        if (k <= 1) return 0; // Edge case

        int n = nums.length;
        int left = 0, prod = 1, count = 0;

        for (int right = 0; right < n; right++) {
            prod *= nums[right];

            // Shrink window till product < k
            while (prod >= k && left <= right) {
                prod /= nums[left];
                left++;
            }

            // All subarrays ending at right with product < k
            count += (right - left + 1);
        }

        return count;
    }
}

    class Solution { //DSAConceptRevision
    /* Function to find the length of the
    longest substring with at most k zeros */
    public int longestOnes(int[] nums, int k) {
        
        int i = 0, zeros = 0, ans = 0;
        
        for (int j = 0; j < nums.length; j++) {
            
            if (nums[j] == 0) zeros++;
            
            // jab zeros zyada ho jaaye allowed se, to window ko shrink karo
            while (zeros > k) {
                if (nums[i] == 0) zeros--;
                i++;
            }
            
            if(zeros <= k) ans = Math.max(ans, j - i + 1);
        }
        
        return ans;
    }
}
  • Minimum Size Subarray Sum (Smallest window with sum ≥ target)
    class Solution {
    public int minSubArrayLen(int target, int[] nums) {
        int n = nums.length;
        int minLen = Integer.MAX_VALUE;
        int left = 0, sum = 0;

        for (int right = 0; right < n; right++) {
            sum += nums[right];

            // Shrink window till sum >= target
            while (sum >= target) {
                minLen = Math.min(minLen, right - left + 1);
                sum -= nums[left];
                left++;
            }
        }

        // If no valid subarray found, return 0
        return (minLen == Integer.MAX_VALUE) ? 0 : minLen;
    }
}

  • Find All Anagrams in a String
    class Solution {
    public List<Integer> findAnagrams(String txt, String pat) {
        int k = pat.length();
        int n = txt.length();
        List<Integer> ans = new ArrayList<>(); // Stores start indices of all anagrams

        if (n < k) return ans;

        // Frequency map for pattern
        Map<Character, Integer> map = new HashMap<>();
        for (int i = 0; i < k; i++) {
            char ch = pat.charAt(i);
            map.put(ch, map.getOrDefault(ch, 0) + 1);
        }

        int count = map.size(); // Unique characters yet to match
        int i = 0; // Left pointer

        for (int j = 0; j < n; j++) {
            char ch = txt.charAt(j);

            // If current char is in pattern, decrement its count
            if (map.containsKey(ch)) {
                map.put(ch, map.get(ch) - 1);
                if (map.get(ch) == 0) count--;
            }

            // Window size matches pattern length
            if (j - i + 1 == k) {
                if (count == 0) ans.add(i); // Anagram found, add start index

                // Move window: restore count of outgoing character
                char out = txt.charAt(i);
                if (map.containsKey(out)) {
                    map.put(out, map.get(out) + 1);
                    if (map.get(out) == 1) count++;
                }
                i++;
            }
        }
        return ans;
    }
}

    }

  • Count number of Nice subarrays (Exactly k odd numbers, use atMost(k) - atMost(k-1))
    class Solution { //DSAQuickRevision
		// Prerequisite: Binary Subarrays With Sum
		// count of subarrays with ≤ k odds - count of subarrays with ≤ k-1 odds
    public int numberOfOddSubarrays(int[] nums, int k) {
        return countSubarraysWithAtMostKOdds(nums, k) 
             - countSubarraysWithAtMostKOdds(nums, k - 1);
    }

    // Helper: Count subarrays with at most k odd numbers
    private int countSubarraysWithAtMostKOdds(int[] nums, int k) {
        
        if (k < 0) return 0; // No subarray if k is negative
        
        int l = 0, count = 0, oddCount = 0;
        
        for (int r = 0; r < nums.length; r++) {
            // Add current element's oddness
            oddCount += nums[r] % 2;

            // Shrink window if oddCount > k
            while (oddCount > k) {
                oddCount -= nums[l] % 2;
                l++;
            }

            // All subarrays ending at r are valid
            count += (r - l + 1);
        }
        
        return count;
    }
}
  • Count Subarrays with K Different Integers (Exactly k distinct, atMost trick)
    class Solution { //DSAStriver

    // Main: Find number of subarrays with exactly k distinct elements
    public int subarraysWithKDistinct(int[] nums, int k) {
        return atMostKDistinct(nums, k) - atMostKDistinct(nums, k - 1);
    }

    // Helper: Find number of subarrays with at most k distinct elements
    private int atMostKDistinct(int[] nums, int k) {
        if (k < 0) return 0; // No valid subarray if k is negative
        
        HashMap<Integer, Integer> map = new HashMap<>();
        int l = 0, count = 0;
        
        for (int r = 0; r < nums.length; r++) {
            // Add nums[r] to window
            map.put(nums[r], map.getOrDefault(nums[r], 0) + 1);
            
            // Shrink window if distinct elements > k
            while (map.size() > k) {
                map.put(nums[l], map.get(nums[l]) - 1);
                if (map.get(nums[l]) == 0) map.remove(nums[l]);
                l++;
            }
            
            // All subarrays ending at r are valid
            count += (r - l + 1);
        }
        
        return count;
    }
}

    class Solution { // DSAConceptRevision
    // Max length subarray with at most 2 types of numbers
    public int totalFruits(int[] fruits) {
        
        int n = fruits.length;
        int maxLen = 0;  
        
        // Map to track the count of each fruit type in the current window
        HashMap<Integer, Integer> mpp = new HashMap<>();
        
        for (int i = 0, j = 0; j < n; j++) {
            mpp.put(fruits[j], mpp.getOrDefault(fruits[j], 0) + 1);
            
            // If the number of different fruits exceeds 2, shrink the window from left
            while (mpp.size() > 2) {
                mpp.put(fruits[i], mpp.get(fruits[i]) - 1);
                if (mpp.get(fruits[i]) == 0) {
                    mpp.remove(fruits[i]);
                }
                i++;
            }
            
            // Update maxLen if the number of different fruits is at most 2
            if(mpp.size() <= 2) maxLen = Math.max(maxLen, j - i + 1);
        }

        return maxLen;
    }
}

                result += n - right;

                if (map.get(s.charAt(left)) == 0) {
                    map.remove(s.charAt(left));
                }
                left++;
            }
        }
        
        return result;
    }

  • Maximum Points You Can Obtain from Cards
    class Solution { //DSAConceptRevision
    /* Function to calculate the maximum
    score after picking k cards */
    public int maxScore(int[] cardScore, int k) {
        int lSum = 0, rSum = 0, maxSum = 0;

        // Calculate the initial sum of the first k cards
        for (int i = 0; i < k; i++) {
            lSum += cardScore[i];
            
            /* Initialize maxSum with the
            sum of the first k cards */
            maxSum = lSum;
        }

        // Initialize rightIndex to iterate array from last
        int rightIndex = cardScore.length - 1;
        
        for (int i = k - 1; i >= 0; i--) {
            
            // Remove the score of the ith card from left sum
            lSum -= cardScore[i];   
            
            /* Add the score of the card
            from the right to the right sum */
            rSum += cardScore[rightIndex];  
            
            // Move to the next card from the right
            rightIndex--; 

            // Update maxSum with the maximum sum found so far
            maxSum = Math.max(maxSum, lSum + rSum);
        }

        // Return the maximum score found
        return maxSum; 
    }
}
  • Binary Subarrays With Sum
    class Solution { //DSAConceptRevision

    // Main function: find number of subarrays with sum exactly equal to 'goal'
    public int numSubarraysWithSum(int[] nums, int goal) {
        // Difference of (sum <= goal) and (sum <= goal-1) gives (sum == goal)
        return numSubarraysWithSumLessEqualToGoal(nums, goal) 
             - numSubarraysWithSumLessEqualToGoal(nums, goal - 1);
    }
    
    // Helper function: find number of subarrays with sum <= goal
    private int numSubarraysWithSumLessEqualToGoal(int[] nums, int goal) {
        
        // Base case: If goal < 0, no subarray possible
        if (goal < 0) return 0;
        
        int l = 0;      // Left pointer
        int sum = 0;    // Current window sum
        int count = 0;  // Total valid subarrays

        for (int r = 0; r < nums.length; r++) {
            sum += nums[r]; // Expand window by adding nums[r]

            // Shrink window from left until sum <= goal
            while (sum > goal) {
                sum -= nums[l];
                l++;
            }

            // For current r, all subarrays [l...r], [l+1...r], ..., [r...r] are valid
            count += (r - l + 1);
        }
        
        return count;
    }
}

  • First negative in every window of size k
    class Solution { //DSAStriver

    // Function to find the first negative integer in every window of size k
    static List<Integer> FirstNegativeInteger(int arr[], int k) {
        // List to store the result
        List<Integer> result = new ArrayList<>();
        // LinkedList to store negative elements in the current window
        LinkedList<Integer> linkedList = new LinkedList<>();
        
        int i = 0; // Start of the window
        
        for (int j = 0; j < arr.length; j++) {
            // Step 1: Add current negative number to the LinkedList
            if (arr[j] < 0) {
                linkedList.addLast(arr[j]);
            }

            // Step 2: Check if the window size is k
            if (j - i + 1 == k) {
                // Step 3: Add the first negative number in the window to the result
                if (linkedList.isEmpty()) {
                    result.add(0); // No negative number in the window
                } else {
                    result.add(linkedList.peekFirst()); // Add the first negative number
                }

                // Step 4: Remove the element that is sliding out of the window
                if (!linkedList.isEmpty() && arr[i] == linkedList.peekFirst()) {
                    linkedList.pollFirst(); // Remove from the front
                }

                // Slide the window
                i++;
            }
        }
        return result;
    }
}

    class Solution { //DSAStriver
    public int findSubString(String str) {
        int minLength = Integer.MAX_VALUE;
        int n = str.length();
        
        // Step 1: Count unique characters in full string
        Set<Character> set = new HashSet<>();
        for (int i = 0; i < n; i++) {
            set.add(str.charAt(i));
        }
        int totalUnique = set.size();

        Map<Character, Integer> map = new HashMap<>();
        int i = 0;

        for (int j = 0; j < n; j++) {
            char ch = str.charAt(j);
            map.put(ch, map.getOrDefault(ch, 0) + 1);

            // Shrink window from left while it's valid
            while (map.size() == totalUnique) {
                minLength = Math.min(minLength, j - i + 1);
                
                char leftChar = str.charAt(i);
                map.put(leftChar, map.get(leftChar) - 1);
                if (map.get(leftChar) == 0) map.remove(leftChar);
                i++;
            }
        }

        return minLength;
    }
