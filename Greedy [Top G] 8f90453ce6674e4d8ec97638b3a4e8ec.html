    class Solution { //DSAStriver
    public boolean canJump(int[] nums) {
        int maxReach = 0; // Maximum index we can reach so far

        for (int i = 0; i < nums.length; i++) {
            if (i > maxReach) return false; // If current index is not reachable

            // Update maxReach based on jump from current index
            maxReach = Math.max(maxReach, i + nums[i]);
        }

        return true; // If loop completes, last index is reachable
    }
}
  • Jump Game 2
    class Solution { //DSAStriver
    public int jump(int[] nums) {
        int jumps = 0;
        int l = 0, r = 0;

        while (r < nums.length - 1) {
            int farthest = 0;

            // Explore all positions in current jump range
            for (int i = l; i <= r; i++) {
                farthest = Math.max(farthest, i + nums[i]);
            }

            // Move to next jump window
            l = r + 1;
            r = farthest;
            jumps++; // One jump taken
        }

        return jumps;
    }
}

    class Solution { //DSAConceptRevision
    public int candy(int[] ratings) {
        int n = ratings.length;

        int[] candies = new int[n];
        Arrays.fill(candies, 1); // Each child gets at least 1 candy

        // 🔼 Left to Right: if right child > left child → give more candy
        for (int i = 1; i < n; i++) {
            if (ratings[i] > ratings[i - 1])
                candies[i] = candies[i - 1] + 1;
        }

        // 🔽 Right to Left: if left child > right child → ensure more candy
        for (int i = n - 2; i >= 0; i--) {
            if (ratings[i] > ratings[i + 1])
                candies[i] = Math.max(candies[i], candies[i + 1] + 1);
        }

        // 🔢 Total candies required
        int total = 0;
        for (int c : candies) total += c;

        return total;
    }
}

  • Non-overlapping Intervals
    class Solution { //DSAConceptRevision
    public int MaximumNonOverlappingIntervals(int[][] intervals) {
        // Sort by end time to maximize non-overlapping intervals
        Arrays.sort(intervals, (a, b) -> Integer.compare(a[1], b[1]));

        int count = 1;  // First interval is always selected
        int lastEndTime = intervals[0][1];

        for (int i = 1; i < intervals.length; i++) {
            // If current interval doesn't overlap, select it
            if (intervals[i][0] >= lastEndTime) {
                count++;
                lastEndTime = intervals[i][1];
            }
        }

        // Minimum to remove = total - non-overlapping count
        return intervals.length - count;
    }
}
  • Valid Parenthesis String
    class Solution { //DSAConceptRevision
    public boolean isValid(String s) {
        int minOpen = 0, maxOpen = 0;

        for (char c : s.toCharArray()) {
            if (c == '(') {
                // Must be treated as open bracket
                minOpen++;  // Minimum opens needed
                maxOpen++;  // Maximum opens possible
            } else if (c == ')') {
                // Must be treated as close bracket
                minOpen--;  // Reduce min open (close one)
                maxOpen--;  // Reduce max open (close one)
            } else if (c == '*') {
                // '*' can be '(', ')' or ''
                minOpen--;  // If '*' = ')'
                maxOpen++;  // If '*' = '('
            }

            // If at any point, more ')' than '(' possible → invalid
            if (maxOpen < 0) return false;

            // minOpen should not be negative
            if (minOpen < 0) minOpen = 0;
        }

        // minOpen == 0 means we found a valid configuration
        return minOpen == 0;
    }
}

  • N meetings in one room
    class Solution { // DSAQuickRevision

    // Sort meetings by end time (ascending)
    static class MeetingComparator implements Comparator<int[]> {
        public int compare(int[] a, int[] b) {
            return Integer.compare(a[1], b[1]);
        }
    }

    public int maxMeetings(int[] start, int[] end) {
        int n = start.length;
        List<int[]> meetings = new ArrayList<>();

        // Create (start, end) meeting pairs
        for (int i = 0; i < n; i++) {
            meetings.add(new int[]{start[i], end[i]});
        }

        // Sort meetings by end time
        Collections.sort(meetings, new MeetingComparator());

        int lastEndTime = meetings.get(0)[1];
        int maxMeetingsCount = 1;

        // Select non-overlapping meetings
        for (int i = 1; i < n; i++) {
            int currStart = meetings.get(i)[0];
            int currEnd = meetings.get(i)[1];
            
            if (currStart > lastEndTime) {
                lastEndTime = currEnd;
                maxMeetingsCount++;
            }
        }

        return maxMeetingsCount;
    }
}
  • Minimum number of platforms required for a railway
    class Solution { //DSAQuickRevision
//  To find number of platforms
public int findPlatform(int[] Arrival, int[] Departure) {
    int n = Arrival.length;

    // Sort both arrival and departure arrays
    Arrays.sort(Arrival);
    Arrays.sort(Departure);

    int ans = 1;
    int count = 1;
    int i = 1, j = 0;

    // Iterate through the arrays
    while (i < n && j < n) {
        if (Arrival[i] <= Departure[j]) {
            // Increment count
            count++;
            i++;
        } else {
            // Decrement count
            count--;
            j++;
        }
        // Find maximum
        ans = Math.max(ans, count);
    }
    return ans;
}
}
  • Job Sequencing problem
    class Solution { //DSAQuickRevision
    // Function to calculate maximum profit
    public int[] JobScheduling(int[][] Jobs) {
        // Sort jobs based on profit in descending order
        Arrays.sort(Jobs, (a, b) -> b[2] - a[2]);

        // Total number of jobs
        int n = Jobs.length;

        /*Initialize a hash table 
        to store selected jobs.
        each element represents a 
        deadline slot, 
        initially unoccupied.*/
        int[] hash = new int[n];
        Arrays.fill(hash, -1);

        // Initialize count
        int cnt = 0;

        // Initialize the total profit earned
        int totalProfit = 0;

        // Iterate over each job
        for (int i = 0; i < n; i++) {
            /*Iterate over each deadline slot starting 
            from the job's deadline*/
            for (int j = Jobs[i][1] - 1; j >= 0; j--) {
                /*If the current deadline 
                slot is available 
                (not occupied)*/
                if (hash[j] == -1) {
                    // Count of selected jobs
                    cnt++;
                    // Mark the job as selected
                    hash[j] = Jobs[i][0];
                    // Update the total profit
                    totalProfit += Jobs[i][2];
                    // Move to the next job
                    break;
                }
            }
        }

        // Return the array
        return new int[]{cnt,totalProfit};
    }
}

  • Fractional Knapsack
    class Solution { //DSAConceptRevision
    double fractionalKnapsack(int[] values, int[] weights, int W) {
        int n = values.length;
        List<int[]> list = new ArrayList<>();

        // Combine value and weight into one object
        for (int i = 0; i < n; i++) {
            list.add(new int[]{values[i], weights[i]});
        }

        // Sort by value/weight ratio in descending order
        Collections.sort(list, (a, b) -> Double.compare((double)b[0]/b[1], (double)a[0]/a[1]));

        double ans = 0;

        // Pick items greedily
        for (int i = 0; i < n; i++) {
            int val = list.get(i)[0];
            int wei = list.get(i)[1];

            if (wei <= W) {
                ans += val;
                W -= wei;
            } else {
                ans += ((double) val / wei) * W; // Take fraction
                break;
            }
        }

        return ans;
    }
}

    class Solution { //DSAStriver
    public int findMaximumCookieStudents(int[] Student, int[] Cookie) {
        int n = Student.length;
        int m = Cookie.length;
        int count = 0;
        int l = 0, r = 0;
        // Sorting of arrays
        Arrays.sort(Student);
        Arrays.sort(Cookie);

        // Traverse through both arrays
        while (l < n && r < m) {
            /*If the current cookie can satisfy 
            the current student, move to the 
            next student*/
            if (Cookie[r] >= Student[l]) {
                l++;
                count++;
            }
            // Move to next cookie
            r++;
        }
        
        return count; 
    }
}
  • Insert Interval
    class Solution { //DSAStriver
    public int[][] insert(int[][] intervals, int[] newInterval) {
        List<int[]> result = new ArrayList<>();

        int i = 0;
        int n = intervals.length;

        // Step 1: Add all intervals before newInterval (no overlap)
        while (i < n && intervals[i][1] < newInterval[0]) {
            result.add(intervals[i]);
            i++;
        }

        // Step 2: Merge all overlapping intervals with newInterval
        while (i < n && intervals[i][0] <= newInterval[1]) {
            newInterval[0] = Math.min(newInterval[0], intervals[i][0]);
            newInterval[1] = Math.max(newInterval[1], intervals[i][1]);
            i++;
        }
        result.add(newInterval); // Add the merged interval

        // Step 3: Add all intervals after newInterval (no overlap)
        while (i < n) {
            result.add(intervals[i]);
            i++;
        }

        // Step 4: Convert List to int[][]
        int[][] ans = new int[result.size()][2];
        for (int j = 0; j < result.size(); j++) {
            ans[j][0] = result.get(j)[0];
            ans[j][1] = result.get(j)[1];
        }

        return ans;
    }
}
  • Lemonade Change
    class Solution { //DSAStriver
    public boolean lemonadeChange(int[] bills) {
        int five = 0, ten = 0;

        for (int bill : bills) {
            if (bill == 5) {
                five++;
            } else if (bill == 10) {
                if (five == 0) {
                    return false; // Can't give change for $10
                }
                five--;
                ten++;
            } else { // bill == 20
                // Important: Prioritize using one $10 bill and one $5 bill for $20 change
                if (ten > 0 && five > 0) {
                    ten--;
                    five--;
                } 
                // If no $10 bill, then use three $5 bills for $20 change
                else if (five >= 3) {
                    five -= 3;
                } 
                // If neither option is available, return false
                else {
                    return false; // Can't give change for $20
                }
            }
        }
        return true;
    }
}
  • Shortest Job first
    class Solution { //DSAStriver
    /* Function to calculate total waiting 
       time using Shortest Job First algorithm */
    public long solve(int[] jobs) {
        // Sort jobs in ascending order
        Arrays.sort(jobs);

        // Initialize total waiting time
        long waitTime = 0;
        // Initialize total time taken
        long totalTime = 0;
        // Get number of jobs
        int n = jobs.length;

        // Iterate to calculate waiting time
        for (int i = 0; i < n; ++i) {
            waitTime += totalTime;
            totalTime += jobs[i];
        }

        // Return average waiting time
        return waitTime / n;
    }
}
